    function playOscillatorFallback() {

        try {

            const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

            const osc = audioCtx.createOscillator();

            const gain = audioCtx.createGain();

            osc.type = 'sine';

            osc.frequency.setValueAtTime(600, audioCtx.currentTime);

            gain.gain.setValueAtTime(0.1, audioCtx.currentTime);

            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.05);

            osc.connect(gain);

            gain.connect(audioCtx.destination);

            osc.start();

            osc.stop(audioCtx.currentTime + 0.05);

        } catch(e) {}

    }

    function recalculateKingOfTheHill() {

        stats = Array(MAX_P).fill(0);

       

        archive.forEach(g => {

            let winText = g.w ? g.w.trim().toLowerCase() : "";

            if (winText === "") return;

            if (winText.startsWith("unentschieden:")) {

                let winnersList = g.w.replace(/Unentschieden:\s*/i, "").split(",");

                winnersList.forEach(wName => {

                    let sName = wName.trim().toLowerCase();

                    let pIdx = pNames.findIndex(name => name && name.trim().toLowerCase() === sName);

                    if (pIdx !== -1) { stats[pIdx]++; }

                });

                return;

            }

           

            if (winText.includes("aggi") || winText.includes("spieler 1")) {

                stats[0]++;

                return;

            }

            if (winText.includes("david") || winText.includes("spieler 2")) {

                stats[1]++;

                return;

            }

           

            if (winText.startsWith("sieger: ")) {

                let sName = g.w.replace(/Sieger:\s*/i, "").trim().toLowerCase();

                let pIdx = pNames.findIndex(name => name && name.trim().toLowerCase() === sName);

                if (pIdx !== -1) { stats[pIdx]++; return; }

            }

            if (g.s && Array.isArray(g.s)) {

                let slotTotals = Array(MAX_P).fill(0);

                g.s.forEach(row => {

                    for (let i = 0; i < MAX_P; i++) {

                        let val = parseInt(row[i]);

                        if (!isNaN(val)) slotTotals[i] += val;

                    }

                });

                let maxScore = Math.max(...slotTotals);

                if (maxScore >= 10000) {

                    for (let i = 0; i < MAX_P; i++) {

                        if (slotTotals[i] === maxScore) {

                            stats[i]++;

                        }

                    }

                }

            }

        });

    }

    function renderRaceTrack() {

        if (!showRace) return;

        const sorted = getSortedActive();

        const activeScores = isArcView ? arcScores : scores;

        sorted.forEach(p => {

            let total = 0;

            activeScores.forEach(r => {

                const v = parseInt(r[p.i]);

                if(!isNaN(v)) total += v;

            });

            let pct = Math.min(100, (total / 10000) * 100);

            const origPath = document.querySelector(`#raceTrackUI #horse-wrapper-${p.i}`)?.closest('.race-track-path');

            const wrapperOrig = document.querySelector(`#raceTrackUI #horse-wrapper-${p.i}`);

            if (origPath && wrapperOrig && textOrig) {

                const pathWidthOrig = origPath.getBoundingClientRect().width;

                const wrapperWidthOrig = wrapperOrig.getBoundingClientRect().width || 65;

                const availableWidthOrig = pathWidthOrig - wrapperWidthOrig;

                const calculatedLeftOrig = (pct / 100) * availableWidthOrig;

                wrapperOrig.style.right = 'auto';

                wrapperOrig.style.left = `${calculatedLeftOrig}px`;

                textOrig.innerText = `${total}`;

            }

            const copyPath = document.querySelector(`#raceTrackStickyCopy #horse-wrapper-${p.i}`)?.closest('.race-track-path');

            const wrapperCopy = document.querySelector(`#raceTrackStickyCopy #horse-wrapper-${p.i}`);

            const textCopy = document.querySelector(`#raceTrackStickyCopy #horse-text-${p.i}`);

            if (copyPath && wrapperCopy && textCopy) {

                const pathWidthCopy = copyPath.getBoundingClientRect().width;

                const wrapperWidthCopy = wrapperCopy.getBoundingClientRect().width || 65;

                const availableWidthCopy = pathWidthCopy - wrapperWidthCopy;

                const calculatedLeftCopy = (pct / 100) * availableWidthCopy;

                wrapperCopy.style.right = 'auto';

                wrapperCopy.style.left = `${calculatedLeftCopy}px`;

                textCopy.innerText = `${total}`;

            }

        });

    }

    function updateUI(sorted) {

        const activeScores = isArcView ? arcScores : scores;

        let dist = { ronny:{}, small:{}, med:{}, big:{}, elite:{}, meister:{} };

        let pendingWinners = [];

        sorted.forEach(p => {

            let total = 0;

            activeScores.forEach(r => {

                const v = parseInt(r[p.i]);

                if(!isNaN(v)) total += v;

            });

            if (total >= 10000) {

                pendingWinners.push(p.i);

            }

        });

        const f = document.getElementById('tRow'); f.innerHTML = '<td>Σ</td>';

        sorted.forEach(p => {

            let total = 0;

            activeScores.forEach(r => {

                const v = parseInt(r[p.i]);

                if(!isNaN(v)) {

                    total += v;

                    let cat = v===0?'ronny':v<=450?'small':v<=700?'med':v<=950?'big':v<=1500?'elite':'meister';

                    if(!dist[cat][p.i]) dist[cat][p.i] = 0;

                    dist[cat][p.i]++;

                }

            });

            let rest = 10000 - total;

           

            if (!isArcView) {

                if (rest < 1001 && rest > 0) {

                    if (!playedUnder1000[p.i]) {

                        playCustomAudio('unter-1000.mp3');

                        playedUnder1000[p.i] = true;

                    }

                } else if (rest >= 1001) {

                    playedUnder1000[p.i] = false;

                }

                if (rest <= 0) {

                    if (!playedWinnerSound[p.i]) {

                        playCustomAudio('sieger.mp3');

                        playedWinnerSound[p.i] = true;

                    }

                } else {

                    playedWinnerSound[p.i] = false;

                }

            }

            let statusClass = "";

            let displayText = rest;

            if (rest <= 0) {

                statusClass = "winner-blink";

                displayText = `👑 ${rest} 👑`;

            } else if (rest <= 1000) {

                statusClass = "near-win";

            }

            f.innerHTML += `<td><div class="score-box ${statusClass}"><span class="score-mobile-name">${p.n}</span><span class="score-down">${displayText}</span><span class="score-up">${total}</span></div></td>`;

        });

        const displayNamesSource = isArcView ? arcNames : pNames;

        document.getElementById('statsDisp').innerHTML = displayNamesSource.map((n,i)=>({n,i})).map(p => {

            if (!p.n || p.n.trim() === "") {

                return `<div class="stat-item" style="opacity: 0.2;"><span class="stat-name">-</span><span class="stat-wins">👑 0</span></div>`;

            }

            let isWinningNow = pendingWinners.includes(p.i) ? '<span class="pending-win">(+1)</span>' : '';

            return `<div class="stat-item"><span class="stat-name">${p.n}</span><span class="stat-wins">👑 ${stats[p.i]||0}${isWinningNow}</span></div>`;

        }).join('');

        const cats = [['RONNYS (0)','ronny'],['350 bis 450','small'],['500 bis 700','med'],['750 bis 950','big'],['1000 bis 1450','elite'],['Über 1500','meister']];

        document.getElementById('analysisUI').innerHTML = cats.map(c => {

            const details = sorted.map(p => `<div class="ana-row" style="font-size:0.9rem; margin-top:3px; display:flex; justify-content:between;"><span>${p.n}:</span> <b style="margin-left:auto;">${dist[c[1]][p.i] || 0}</b></div>`).join('');

            return `<div class="analysis-card"><div style="font-size:0.95rem; font-weight:bold; color:var(--accent); border-bottom: 1px solid var(--border); padding-bottom:3px; margin-bottom:5px;">${c[0]}</div>${details}</div>`;

        }).join('');

        renderRaceTrack();

    }

    function getSortedActive() {

        const names = isArcView ? arcNames : pNames;

        const active = names.map((n, i) => ({n, i})).filter(p => p.n && p.n.trim() !== "");

        if (active.length === 0) return [];

        const sPos = active.findIndex(p => p.i === startIdx);

        return (sPos === -1) ? active : [...active.slice(sPos), ...active.slice(0, sPos)];

    }

   

    async function saveCSV() {

        const d = { pNames, scores, archive, startIdx, pEmojis };

        const jsonString = JSON.stringify(d);

        const now = new Date();

        const yy = String(now.getFullYear()).slice(-2);

        const mm = String(now.getMonth() + 1).padStart(2, '0');

        const dd = String(now.getDate()).padStart(2, '0');

        const hh = String(now.getHours()).padStart(2, '0');

        const min = String(now.getMinutes()).padStart(2, '0');

        const fileName = `Würfel-Backup-${yy}.${mm}.${dd}-${hh}.${min}.csv`;

       

        try {

            const blob = new Blob([jsonString], { type: 'text/csv;charset=utf-8;' });

            const a = document.createElement('a');

            a.href = URL.createObjectURL(blob);

            a.download = fileName;

            document.body.appendChild(a);

            a.click();

            document.body.removeChild(a);

           

            openAlert("Erfolg", "Das Backup wurde erfolgreich erstellt und der Download wurde gestartet.");

        } catch (err) {

            openAlert("Fehler", "Das Backup konnte nicht erstellt werden.");

        }

    }

    function loadCSV(e) {

        const r = new FileReader();

        r.onload = (ev) => {

            try {

                const d = JSON.parse(ev.target.result);

                pNames=d.pNames; scores=d.scores; archive=d.archive; startIdx=d.startIdx;

                if(d.pEmojis) pEmojis=d.pEmojis;

                save(); render();

            } catch(err) {

                openAlert("Fehler", "Die Datei konnte nicht als gültiges Backup geladen werden!");

            }

        };

        r.readAsText(e.target.files[0]);

    }
